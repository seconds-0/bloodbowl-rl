import json,glob,sys,statistics as st
def load(pat):
    f=[p for p in glob.glob(pat) if p.endswith(".log")][0]
    rows=[]
    for line in open(f,errors="replace"):
        if line.startswith("PUFFER_LOSS_JSON "):
            try: d=json.loads(line[len("PUFFER_LOSS_JSON "):])
            except Exception: continue
            if d.get("_puffer_phase_eval")==0 and d.get("_puffer_epoch",-1)>=0 and not d.get("_puffer_final_reprint"):
                rows.append(d)
    return rows
